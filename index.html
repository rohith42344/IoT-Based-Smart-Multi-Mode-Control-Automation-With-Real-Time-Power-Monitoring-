One ESP32 switches five loads from seven control modes, measures voltage, current, power and energy, and reports to two cloud dashboards. Built at the Government Tool Room and Training Centre (GTTC), Magadi.

Project overview

Problem

Manual switching wastes energy, gives no usage data and offers no remote protection against overload.

Solution

An ESP32 controller combines physical switches, mobile apps, SMS, an IR remote and Bluetooth, with live energy monitoring on an LCD and in the cloud.

Applications

Smart homes, industrial load control, energy management systems and cloud data logging.

IoT dashboard creation

Blynk IoT

  1. Create a template named HOME AUTOMATION and add the datastreams below.
  2. Add the template ID, template name and auth token to the top of the sketch.
  3. Build the web and mobile dashboard: five switches (V1–V5), two buttons (V12, V13), gauges for V6–V8, V11, V14, and value displays for V9 and V10.
  4. Add a device from the template; the sketch connects with Blynk.config().

ThingsBoard Cloud

  1. Sign in at thingsboard.cloud, add a device and copy its access token.
  2. Put that token in TB_TOKEN. The device connects to mqtt.thingsboard.cloud:1883 with the token as the user name and an empty password.
  3. Create a dashboard and add cards for the telemetry keys: voltage, current, power, energy, bill, temperature, humidity, relay1…relay5.
  4. For each relay add a Switch control widget using RPC method setRelay, with the parameter built as shown below.

2. Relay control (active low) and Blynk handlers 
void setRelay(uint8_t index, bool on) {
  if (index >= RELAY_COUNT) return;
  relayState[index] = on;
  digitalWrite(relayPins[index], on ? LOW : HIGH);   // ON = LOW
}
void allON()  { for (uint8_t i=0;i<RELAY_COUNT;i++) setRelay(i,true);  }
void allOFF() { for (uint8_t i=0;i<RELAY_COUNT;i++) setRelay(i,false); }

3. ThingsBoard RPC callback 
void tbCallback(char* topic, byte* payload, unsigned int length) {
  String t(topic);
  String requestId = t.substring(t.lastIndexOf('/') + 1);
  StaticJsonDocument<512> req;
  if (deserializeJson(req, payload, length)) return;

  const char* method = req["method"] | "";
  JsonVariant params = req["params"];
  bool success = false;

  if (strcmp(method, "setRelay") == 0) {
    int n = params["relay"] | 0;
    if (n >= 1 && n <= RELAY_COUNT && !params["state"].isNull()) {
      setRelay(n - 1, params["state"].as<bool>());
      success = true;
    }
  } else if (strcmp(method, "setAllRelays") == 0 && !params["state"].isNull()) {
    params["state"].as<bool>() ? allON() : allOFF();
    success = true;
  } else if (strcmp(method, "getRelayStates") == 0) success = true;

  // publish reply to v1/devices/me/rpc/response/<requestId>, then:
  syncAllRelays();
  sendThingsBoardTelemetry();
}
4. Power monitor (PZEM-004T) and DHT11 
void readPowerMonitor() {
  float v = pzem.voltage(), c = pzem.current();
  float w = pzem.power(),   e = pzem.energy();
  pzemVoltageValid = !isnan(v) && v >= 0;
  pzemCurrentValid = !isnan(c) && c >= 0;
  pzemPowerValid   = !isnan(w) && w >= 0;
  pzemEnergyValid  = !isnan(e) && e >= 0;
  if (pzemVoltageValid) voltage = v;
  if (pzemCurrentValid) current = c;
  if (pzemPowerValid)   power   = w;
  if (pzemEnergyValid)  { energy = e; bill = energy * UNIT_COST; } // Rs 10 per kWh
}
void readTemperature() {
  float t = dht.readTemperature(), h = dht.readHumidity();
  if (!isnan(t)) { temperatureC = t; temperatureValid = true; }
  if (!isnan(h)) { humidity = h;     humidityValid = true; }
}
5. Switches, IR remote and command processor 
void scanPhysicalSwitches() {
  if (millis() - lastSwitchScan < 40) return;
  lastSwitchScan = millis();
  for (uint8_t i = 0; i < RELAY_COUNT; i++) {
    bool s = digitalRead(switchPins[i]);
    if (previousSwitchState[i] == HIGH && s == LOW) {   // falling edge
      setRelay(i, !relayState[i]);
      syncAllRelays(); sendThingsBoardTelemetry();
    }
    previousSwitchState[i] = s;
  }
}
// IR codes: 0x45 R1, 0x46 R2, 0x47 R3, 0x44 R4, 0x40 R5,
//           0x16 ALL ON, 0x0D ALL OFF
void processCommand(String cmd) {
  cmd.trim(); cmd.toUpperCase();
  if (cmd == "ALL ON") allON();
  else if (cmd == "ALL OFF") allOFF();
  else if (cmd == "STATUS") { /* build report, send by BT and SMS */ return; }
  else if (cmd.length() >= 5 && cmd.charAt(0)=='R' && isDigit(cmd.charAt(1))) {
    int n = cmd.substring(1,2).toInt();
    String a = cmd.substring(2); a.trim();
    if (n >= 1 && n <= RELAY_COUNT && (a=="ON" || a=="OFF")) setRelay(n-1, a=="ON");
  }
  syncAllRelays(); sendThingsBoardTelemetry();
}
6. Setup and main loop 
void setup() {
  for (uint8_t i=0;i<RELAY_COUNT;i++) { pinMode(relayPins[i],OUTPUT); digitalWrite(relayPins[i],HIGH); }
  pinMode(switchPins[0],INPUT); pinMode(switchPins[1],INPUT);        // external pull-ups
  for (uint8_t i=2;i<RELAY_COUNT;i++) pinMode(switchPins[i],INPUT_PULLUP);
  Wire.begin(21,22); lcd.init(); lcd.backlight();
  dht.begin(); IrReceiver.begin(IR_PIN, DISABLE_LED_FEEDBACK);
  SerialBT.begin("ESP32_HOME");
  PZEMSerial.begin(9600, SERIAL_8N1, PZEM_RX, PZEM_TX);
  GSM.begin(9600, SERIAL_8N1, GSM_RX, GSM_TX);
  GSM.println("AT+CMGF=1"); GSM.println("AT+CNMI=2,2,0,0,0");     // text mode, push SMS
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Blynk.config(BLYNK_AUTH_TOKEN);
  tb.setServer(TB_SERVER, TB_PORT); tb.setCallback(tbCallback);
  timer.setInterval(2000L, readPowerMonitor);
  timer.setInterval(3000L, readTemperature);
  timer.setInterval(5000L, sendBlynkTelemetry);
  timer.setInterval(5000L, sendThingsBoardTelemetry);
}
void loop() {
  if (WiFi.status()==WL_CONNECTED) { if (Blynk.connected()) Blynk.run(); if (tb.connected()) tb.loop(); }
  timer.run(); scanPhysicalSwitches();
  checkBluetoothConnection(); handleBluetooth();
  handleGSM(); handleIR(); updateLCD(); delay(1);
}
